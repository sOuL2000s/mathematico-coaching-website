        // Global variable to hold the current confirm callback
        let currentConfirmCallback = null;

        // Function to show a custom message box
        function showMessageBox(title, message, isConfirm = false, onConfirm = null) {
            messageBoxTitle.textContent = title;
            messageBoxContent.textContent = message;
            messageBoxOkBtn.textContent = isConfirm ? "Yes" : "OK";

            // Clear any previous click handler on the OK button
            if (currentConfirmCallback) {
                messageBoxOkBtn.removeEventListener('click', currentConfirmCallback);
                currentConfirmCallback = null;
            }
            // Ensure the generic hideMessageBox listener is also removed if it was added
            messageBoxOkBtn.removeEventListener('click', hideMessageBox);

            // Remove any existing 'No' button listener and hide it
            const noBtn = document.getElementById('message-box-no-btn');
            if (noBtn) {
                noBtn.removeEventListener('click', hideMessageBox); // Remove its listener
                noBtn.remove(); // Remove element from DOM
            }

            if (isConfirm) {
                currentConfirmCallback = async () => { // Make sure this is async
                    hideMessageBox(); // Hide first
                    if (onConfirm) {
                        await onConfirm(); // Await the async callback
                    }
                };
                messageBoxOkBtn.addEventListener('click', currentConfirmCallback);

                // Add a "No" button for confirmation if it doesn't exist
                let newNoBtn = document.createElement('button');
                newNoBtn.id = 'message-box-no-btn';
                newNoBtn.className = 'bg-gray-400 text-white px-8 py-3 rounded-full text-md font-semibold hover:bg-gray-500 transition duration-300 ease-in-out shadow-md hover:shadow-lg focus:outline-none focus:ring-2 focus:ring-gray-400 ml-2';
                newNoBtn.textContent = 'No';
                messageBoxOkBtn.parentNode.appendChild(newNoBtn);

                newNoBtn.addEventListener('click', hideMessageBox); // No button always just hides
            } else {
                // For non-confirm messages, just hide the box
                messageBoxOkBtn.addEventListener('click', hideMessageBox);
            }
            messageBoxModal.classList.add('active');
        }

            /**
             * Compresses an image using HTML Canvas.
             * @param {File} file The image file to compress.
             * @param {number} maxWidth Maximum width for the compressed image.
             * @param {number} maxHeight Maximum height for the compressed image.
             * @param {number} quality JPEG quality (0 to 1).
             * @returns {Promise<Blob>} A promise that resolves with the compressed image as a Blob.
             */
            function compressImage(file, maxWidth = 1200, maxHeight = 1200, quality = 0.8) {
                return new Promise((resolve, reject) => {
                    const reader = new FileReader();
                    reader.readAsDataURL(file);
                    reader.onload = event => {
                        const img = new Image();
                        img.src = event.target.result;
                        img.onload = () => {
                            let width = img.width;
                            let height = img.height;

                            if (width > height) {
                                if (width > maxWidth) {
                                    height = Math.round(height * (maxWidth / width));
                                    width = maxWidth;
                                }
                            } else {
                                if (height > maxHeight) {
                                    width = Math.round(width * (maxHeight / height));
                                    height = maxHeight;
                                }
                            }

                            const canvas = document.createElement('canvas');
                            canvas.width = width;
                            canvas.height = height;
                            const ctx = canvas.getContext('2d');
                            ctx.drawImage(img, 0, 0, width, height);

                            canvas.toBlob(blob => {
                                resolve(blob);
                            }, 'image/jpeg', quality);
                        };
                        img.onerror = error => reject(error);
                    };
                    reader.onerror = error => reject(error);
                });
            }

            // Function to upload file to Cloudinary
            async function uploadFile(file, folderName, spinnerElement, previewElement) {
                if (!file) {
                    console.warn("No file provided for upload.");
                    return null;
                }

                if (!CLOUDINARY_CLOUD_NAME || !CLOUDINARY_UPLOAD_PRESET) {
                    showMessageBox("Cloudinary Error", "Cloudinary credentials (CLOUD_NAME or UPLOAD_PRESET) are not configured. Please check your script.");
                    console.error("Cloudinary credentials missing.");
                    return null;
                }

                spinnerElement.classList.remove('hidden'); // Show spinner
                if (previewElement) previewElement.classList.add('hidden'); // Hide preview during upload

                try {
                    // Step 1: Compress the image (optional but recommended for bandwidth)
                    const compressedBlob = await compressImage(file);
                    console.log(`Original size: ${file.size} bytes, Compressed size: ${compressedBlob.size} bytes`);

                    // Step 2: Prepare FormData for Cloudinary upload
                    const formData = new FormData();
                    formData.append('file', compressedBlob);
                    formData.append('upload_preset', CLOUDINARY_UPLOAD_PRESET);
                    // Optionally specify a folder in Cloudinary
                    if (folderName) {
                        formData.append('folder', folderName);
                    }
                    // Add filename to metadata if desired, Cloudinary automatically handles public_id if not specified.
                    // formData.append('public_id', file.name.split('.')[0] + '_' + Date.now()); // Custom public ID

                    const response = await fetch(`https://api.cloudinary.com/v1_1/${CLOUDINARY_CLOUD_NAME}/image/upload`, {
                        method: 'POST',
                        body: formData
                    });

                    if (!response.ok) {
                        const errorData = await response.json();
                        throw new Error(`Cloudinary upload failed: ${errorData.error.message || response.statusText}`);
                    }

                    const data = await response.json();
                    const downloadURL = data.secure_url; // Use secure_url for HTTPS

                    spinnerElement.classList.add('hidden');
                    if (previewElement) {
                        previewElement.src = downloadURL;
                        previewElement.classList.remove('hidden');
                    }
                    return downloadURL;

                } catch (error) {
                    spinnerElement.classList.add('hidden');
                    console.error("Image Upload Failed:", error);
                    showMessageBox("Upload Error", `Image upload failed: ${error.message}. Please check your Cloudinary configuration and network.`);
                    return null;
                }
            }

        function renderBlogs(snapshot) {
            blogDisplay.innerHTML = ''; // Clear previous content
            allBlogPosts = []; // Clear previous posts and populate with new data
            const blogPosts = [];
            snapshot.forEach((doc) => {
                blogPosts.push({ id: doc.id, ...doc.data() });
            });

            // Client-side sorting by date_posted in descending order
            blogPosts.sort((a, b) => {
                const dateA = a.date_posted && a.date_posted.seconds ? a.date_posted.seconds : 0;
                const dateB = b.date_posted && b.date_posted.seconds ? b.date_posted.seconds : 0;
                return dateB - dateA;
            });
            allBlogPosts = blogPosts; // Store sorted posts

            if (blogPosts.length === 0) {
                blogDisplay.innerHTML = '<p class="text-gray-600 dark:text-gray-300 text-center col-span-full py-10">No blog posts yet.</p>';
            } else {
                blogPosts.forEach((post) => {
                    const postElement = document.createElement('article');
                    postElement.className = 'bg-white p-6 rounded-2xl shadow-xl hover:shadow-2xl transform hover:translate-y-[-5px] transition duration-300 ease-in-out flex flex-col items-start text-left dark:bg-gray-800 dark:shadow-2xl border border-yellow-100 dark:border-gray-700';
                    postElement.innerHTML = `
                        <h3 class="text-2xl font-bold text-gray-900 mb-3 dark:text-gray-100">${post.title}</h3>
                        ${post.imageUrl ? `<img src="${post.imageUrl}" alt="${post.title}" class="w-full h-48 object-cover rounded-xl mb-4" loading="lazy" onerror="this.onerror=null;this.src='https://placehold.co/600x200/E0E7FF/4F46E5?text=Blog+Image';">` : ''}
                        <p class="text-gray-600 text-base mb-4 dark:text-gray-300 flex-grow">${post.content.substring(0, 180)}...</p>
                        <p class="text-sm text-gray-500 dark:text-gray-400 mb-2">Posted on: ${post.date_posted ? new Date(post.date_posted.seconds * 1000).toLocaleDateString() : 'N/A'}</p>
                        <button class="read-more-blog-btn text-indigo-600 hover:underline dark:text-indigo-400 mt-auto text-sm font-medium focus:outline-none focus:ring-2 focus:ring-indigo-500" data-id="${post.id}" aria-label="Read more about ${post.title}">Read More &rarr;</button>
